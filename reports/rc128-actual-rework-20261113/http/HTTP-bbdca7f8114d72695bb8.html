Describe the current behavior / 问题描述 (Mandatory / 必填)

一句话：aclnnGroupedMatmulFinalizeRoutingV2 在开启确定性计算（aclrtCtxSetSysParamOpt(ACL_OPT_DETERMINISTIC, 1)）后，kernel 侧触发 AI Core 异常：aclrtSynchronizeStream 返回 507015（aicore exception），plog 明确 The GM address accessed by scalar exceeds 48 bits（kernel 里 scalar 计算出的 GM 地址超过 48 位，属非法地址访问）。关闭确定性后，同一参数、同一份代码 3/3 正常。

两段式接口本身"看起来成功"，错误只在同步阶段暴露：

确定性 rc1（GetWorkspaceSize） rc2（Run） aclrtSynchronizeStream
on（det=1） 0 0 507015 [aicore exception]
off（det=0，对照） 0 0 0

必现性（本轮 12/12 + 复验 4/4）

机型 参数集 det=1 det=0
Atlas A2 910B2（npu:5） m=1, k=2048, n=128, e=1 sync=507015 ×3 sync=0 ×3
Atlas A2 910B2 最小 m=1, k=512, n=128, e=1 sync=507015 ×3 sync=0 ×3
Atlas A3 910B3（npu:5） m=1, k=2048, n=128, e=1 sync=507015 ×3 sync=0 ×3
Atlas A3 910B3 最小 m=1, k=512, n=128, e=1 sync=507015 ×3 sync=0 ×3

另于 Atlas A3 910B3 / npu:5 用下附最小 C 探针独立复验 2 次：det=1 → rc1=0 rc2=0 sync=507015；det=0 → rc1=0 rc2=0 sync=0。

workspace 分叉（说明确定性走的是另一套 tiling 分支）

机型 det=1 det=0
Atlas A2 910B2 222,298,624 B 121,635,328 B
Atlas A3 910B3 205,521,408 B 104,858,112 B

定级依据（建议 P0）

  1. 中断型损失：不是数值偏差，而是 AI Core 异常（aicore exception, retCode=0x26）+ 非法 GM 地址（exceeds 48 bits），任务直接失败；
  2. 走官方支持路径：确定性计算是对外承诺的能力（ACL_OPT_DETERMINISTIC=1），用户按官方方式开启即稳定触发；
  3. 必现、3 秒级复现：双机型 12/12；最小参数已缩到 m=1/k=512/n=128/e=1；
  4. 地址异常具备越界风险：fault 是 scalar 算出 >48 位 GM 地址，当前表现为访问异常，同类缺陷在其他 shape 下可能演变为数据踩踏（污染型损失），不宜按 P1 降级。

// probe_det_507015_min.c - minimal, self-contained reproducer for:
//   aclnnGroupedMatmulFinalizeRoutingV2 + aclrtCtxSetSysParamOpt(ACL_OPT_DETERMINISTIC, 1)
//   => aclrtSynchronizeStream returns 507015 [aicore exception],
//      plog: "The GM address accessed by scalar exceeds 48 bits"
//
// Build:
//   source <CANN set_env.sh>
//   gcc -O2 probe_det_507015_min.c -o probe_det_507015_min \
//       -I$ASCEND_HOME_PATH/include -L$ASCEND_HOME_PATH/lib64 -lopapi -lnnopbase -lascendcl
// Run:
//   ./probe_det_507015_min <deviceId> <det:0|1>
//     det=1 -> rc1=0 rc2=0 sync=507015   (defect)
//     det=0 -> rc1=0 rc2=0 sync=0        (control)
//
// Fixed minimal geometry: m=1, k=512, n=128, e=1 (x2 is the INT32 carrier [e,k,n/8]).
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include "acl/acl.h"
#include "aclnn/aclnn_base.h"
#include "aclnnop/aclnn_grouped_matmul_finalize_routing_v2.h"

static aclTensor *mk(const int64_t *d, uint64_t nd, aclDataType t, void *p)
{
    int64_t st[8], s = 1;
    for (int i = (int)nd - 1; i >= 0; --i) { st[i] = s; s *= d[i]; }
    return aclCreateTensor(d, nd, t, st, 0, ACL_FORMAT_ND, d, nd, p);
}

#define M 1
#define K 512
#define N 128
#define E 1

int main(int argc, char **argv)
{
    if (argc < 3) { printf("usage: %s <deviceId> <det:0|1>\n", argv[0]); return 64; }
    int dev = atoi(argv[1]), det = atoi(argv[2]);

    printf("PROBE dev=%d det=%d m=%d k=%d n=%d e=%d\n", dev, det, M, K, N, E);
    printf("aclInit=%d\n", (int)aclInit(NULL));
    printf("setDevice=%d\n", (int)aclrtSetDevice(dev));
    if (det) { printf("setDeterministic=%d\n", (int)aclrtCtxSetSysParamOpt(ACL_OPT_DETERMINISTIC, 1)); }
    aclrtStream stream = NULL;
    printf("createStream=%d\n", (int)aclrtCreateStream(&stream));

    size_t n8 = (size_t)N / 8;
    void *x1 = NULL, *x2 = NULL, *sc = NULL, *bi = NULL, *pt = NULL, *gl = NULL, *lg = NULL, *ri = NULL, *ou = NULL;
    aclrtMalloc(&x1, (size_t)M * K * 1, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&x2, (size_t)E * K * n8 * 4, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&sc, (size_t)E * N * 8, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&bi, (size_t)N * 4, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&pt, (size_t)M * 4, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&gl, (size_t)E * 8, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&lg, (size_t)M * 4, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&ri, (size_t)M * 8, ACL_MEM_MALLOC_HUGE_FIRST);
    aclrtMalloc(&ou, (size_t)M * N * 4, ACL_MEM_MALLOC_HUGE_FIRST);

    int8_t *hx1 = (int8_t *)malloc((size_t)M * K); memset(hx1, 1, (size_t)M * K);
    int32_t *hx2 = (int32_t *)malloc((size_t)E * K * n8 * 4);
    for (size_t i = 0; i < (size_t)E * K * n8; ++i) { hx2[i] = 1; }
    int64_t *hsc = (int64_t *)malloc((size_t)E * N * 8);
    for (size_t i = 0; i < (size_t)E * N; ++i) { hsc[i] = 0x3F800000LL; }   // fp32 bits(1.0) in low 32 bits
    float *hbi = (float *)calloc((size_t)N, 4);
    float *hpt = (float *)malloc((size_t)M * 4); for (int i = 0; i < M; ++i) { hpt[i] = 1.0f; }
    int64_t *hgl = (int64_t *)malloc((size_t)E * 8); for (int i = 0; i < E; ++i) { hgl[i] = (i == E - 1) ? M : (int64_t)((i + 1) * M / E); }
    float *hlg = (float *)calloc((size_t)M, 4);
    int64_t *hri = (int64_t *)malloc((size_t)M * 8); for (int i = 0; i < M; ++i) { hri[i] = i; }

    aclrtMemcpy(x1, (size_t)M * K, hx1, (size_t)M * K, ACL_MEMCPY_HOST_TO_DEVICE);
    aclrtMemcpy(x2, (size_t)E * K * n8 * 4, hx2, (size_t)E * K * n8 * 4, ACL_MEMCPY_HOST_TO_DEVICE);
    aclrtMemcpy(sc, (size_t)E * N * 8, hsc, (size_t)E * N * 8, ACL_MEMCPY_HOST_TO_DEVICE);
    aclrtMemcpy(bi, (size_t)N * 4, hbi, (size_t)N * 4, ACL_MEMCPY_HOST_TO_DEVICE);
    aclrtMemcpy(pt, (size_t)M * 4, hpt, (size_t)M * 4, ACL_MEMCPY_HOST_TO_DEVICE);
    aclrtMemcpy(gl, (size_t)E * 8, hgl, (size_t)E * 8, ACL_MEMCPY_HOST_TO_DEVICE);
    aclrtMemcpy(ri, (size_t)M * 8, hri, (size_t)M * 8, ACL_MEMCPY_HOST_TO_DEVICE);

    int64_t d_x1[2] = {M, K}, d_x2[3] = {E, K, (int64_t)n8}, d_sc[3] = {E, 1, N}, d_bn[2] = {E, N};
    int64_t d_pt[1] = {M}, d_gl[1] = {E}, d_out[2] = {M, N};
    aclTensor *t_x1 = mk(d_x1, 2, ACL_INT8, x1);
    aclTensor *t_x2 = mk(d_x2, 3, ACL_INT32, x2);
    aclTensor *t_sc = mk(d_sc, 3, ACL_INT64, sc);
    aclTensor *t_bi = mk(d_bn, 2, ACL_FLOAT, bi);
    aclTensor *t_pt = mk(d_pt, 1, ACL_FLOAT, pt);
    aclTensor *t_gl = mk(d_gl, 1, ACL_INT64, gl);
    aclTensor *t_lg = mk(d_pt, 1, ACL_FLOAT, lg);
    aclTensor *t_ri = mk(d_pt, 1, ACL_INT64, ri);
    aclTensor *t_out = mk(d_out, 2, ACL_FLOAT, ou);

    uint64_t wsSize = 0;
    aclOpExecutor *exec = NULL;
    aclnnStatus rc1 = aclnnGroupedMatmulFinalizeRoutingV2GetWorkspaceSize(
        t_x1, t_x2, t_sc, t_bi, NULL, NULL, NULL, t_pt, t_gl, NULL, t_lg, t_ri,
        0 /*dtype*/, 1.0f /*sharedInputWeight*/, 0 /*sharedInputOffset*/,
        false /*transposeX1*/, false /*transposeX2*/, 1 /*groupListType*/,
        t_out, &wsSize, &exec);
    printf("RC1=%d wsSize=%llu\n", (int)rc1, (unsigned long long)wsSize);

    int rc2 = -999, rcs = -999;
    if (rc1 == 0) {
        void *ws = NULL;
        if (wsSize > 0) { aclrtMalloc(&ws, wsSize, ACL_MEM_MALLOC_HUGE_FIRST); }
        rc2 = (int)aclnnGroupedMatmulFinalizeRoutingV2(ws, wsSize, exec, stream);
        printf("RC2=%d\n", rc2);
        rcs = (int)aclrtSynchronizeStream(stream);
        printf("SYNC=%d\n", rcs);
        if (ws) { aclrtFree(ws); }
    }
    printf("VERDICT rc1=%d rc2=%d sync=%d\n", (int)rc1, rc2, rcs);

    aclDestroyTensor(t_x1); aclDestroyTensor(t_x2); aclDestroyTensor(t_sc); aclDestroyTensor(t_bi);
    aclDestroyTensor(t_pt); aclDestroyTensor(t_gl); aclDestroyTensor(t_lg); aclDestroyTensor(t_ri); aclDestroyTensor(t_out);
    aclrtFree(x1); aclrtFree(x2); aclrtFree(sc); aclrtFree(bi); aclrtFree(pt); aclrtFree(gl); aclrtFree(lg); aclrtFree(ri); aclrtFree(ou);
    free(hx1); free(hx2); free(hsc); free(hbi); free(hpt); free(hgl); free(hlg); free(hri);
    aclrtDestroyStream(stream); aclrtResetDevice(dev); aclFinalize();
    return (rc1 == 0 && rc2 == 0 && rcs == 0) ? 0 : 2;
}

Special notes for this issue/备注 (Optional / 选填)

与历史反馈的关系（避免重复排查）

  • #4334（已关闭）《910B2 grouped_matmul_finalize_routing算子开确定性之后卡死》：同族同开关（torch.use_deterministic_algorithms(True)），在 CANN 9.0.0 / 910B2 上表现为卡死无返回，单内无修复说明。本单在 CANN 9.2.x 上给出的是可诊断的崩溃形态（507015 + GM 地址超 48 位 + fault kernel 名 + 最小参数），且 A2/A3 双机型一致——若该单已按"卡死"修复，则本单属修复不完整/回归；若未修复，本单给出了根因方向的直接线索。
  • #3397（已关闭）同族同开关，内容是性能劣化（4120us→8420us），与本单（崩溃）不同。
  • #6542 / #6543 / #6544（已关闭）同族 groupList 取负导致的 507015：那是非法入参路径；本单全部入参合法（groupList={1}、rowIndex={0}、scale 按位模式正确编码），唯一变量是确定性开关，因此根因不同。
  • 未在他人 issue 下追加评论。

给定位的线索（不作为结论）

  • 确定性开关使 workspace 从 121MB 涨到 222MB（A2）/ 104MB 涨到 205MB（A3），说明 det 路径用了另一套 tiling/分块；报错是 scalar 算出的 GM 地址超 48 位，且 fixp_error0/1 有具体值，怀疑与 det 分支里某张量的基址/偏移拼接（高位未截断或偏移量溢出）有关。
  • 触发边界 k≥512 且 n=128，k≤448 不复现，提示与分块阈值（K 方向 256 的整数倍）相关。

其他

  • 本轮未生成 device dump（未配置 dump 落盘路径），plog 内仅有 dump info 的 pc 值；如需 dump 我可以按指导补配后重跑。

从您提供的结果看，开启确定性计算后，同步阶段出现了 AI Core 异常，值得沿该路径继续定位；当前尚不能仅凭日志确定具体根因。

可以重点核对确定性路径的分块参数及 kernel 地址计算；若确认该组合不受支持，也可评估在第一段接口明确报错。确定性计算通常关注相同条件下重复运行的结果一致性，不能仅凭开关要求开启和关闭两种模式的结果逐位相同。问题的影响范围、与历史问题的关系及优先级仍需进一步确认。
