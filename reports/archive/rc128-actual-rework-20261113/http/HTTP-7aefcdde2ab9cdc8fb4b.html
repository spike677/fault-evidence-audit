[Bug-Report|缺陷反馈]: aclnnEmbeddingBag 空 indices 早退返回 SUCCESS，非空 output 保持未初始化 #8389
liuhanwen8Add on  4 days ago
liuhanwen8
4 days ago Create
4 days ago Create

关联：#704（kernel 侧空 tensor 处理，已关闭；本条是 host 侧早退，未到 kernel）

现象

aclnnEmbeddingBag 在 indices 为空（0 元素）时直接早退返回 SUCCESS，但 output 是非空的 (numBags, D) 张量，整块保持未初始化。

实测 weight=[4,8]（第 i 行全为 i+1）、indices=[]（空）、offsets=[0]、mode=0(sum)、includeLastOffset=false：

组 indices offsets output 两段返回 实测 output[0]
对照组 4 个（0,1,2,3） [0,2] (2,8) 0/0 10.0（= weight[0]+weight[1]）
缺陷组 0 个 [0] (1,8) 0/0 保持 canary 3.39615e+38
缺陷组 0 个 [0,0] (2,8) 0/0 保持 canary 3.39615e+38

源码侧（cann/ops-nn master d1efef7c30ca124dc30787a7db3c4479c8ffcc05）index/embedding_bag/op_api/aclnn_embedding_bag.cpp：CheckParams（:344-367）只校验空指针、dtype、维度与 CheckShape（:258），不涉及「输入为空、输出非空」这一组合；随后 :414-418 是 if (weight->IsEmpty() || indices->IsEmpty() || offsets->IsEmpty()) { *workspaceSize = 0; uniqueExecutor.ReleaseTo(executor); return ACLNN_SUCCESS; }，分支内没有任何写 output 的算子。

资料 index/embedding_bag/docs/aclnnEmbeddingBag.md：output 为 2 维、dtype 与 weight 一致（:241-248），形状由 bag 数决定；indices 支持 0-2 维（:161-168）；错误码表 :332-341 的 161002 只覆盖 dtype 不在范围与「维度不是 1 维」，没有「空 tensor 时不写输出、直接返回」的声明。

期望

offsets 非空即确定了 bag 数与 output 形状；空 bag 的 sum 聚合结果应为 0（PyTorch F.embedding_bag(weight, empty_indices, offsets, mode='sum') 返回全 0 的 (numBags, D)）。因此 output 应写 0；若认为空 indices 非法，应返回 ACLNN_ERR_PARAM_INVALID。两种处理都不同于「返回 SUCCESS 且输出保持未初始化」。

环境

Atlas A2 训练服务器（910B2）；CANN 9.3.0（/usr/local/Ascend/ascend-toolkit/latest），Linux aarch64；cann/ops-nn master d1efef7c30ca124dc30787a7db3c4479c8ffcc05；复现日期 2026-11-08。

复现步骤

  1. 保存下方脚本为 repro.cpp，编译：g++ -std=c++17 -O2 repro.cpp -o repro -I$ASCEND_HOME_PATH/include -L$ASCEND_HOME_PATH/lib64 -lascendcl -lnnopbase -lopapi -Wl,--allow-shlib-undefined，运行 ./repro：对照组 output 首元素为 10.0；两个缺陷组 output 整块保持 0x7F 预填值

复现脚本

#include <cstdio>
#include <cstring>
#include <vector>
#include "acl/acl.h"
#include "aclnnop/aclnn_embedding_bag.h"
static void* Buf(size_t n, unsigned char f)
{
    size_t m = n ? n : 32; void* p = nullptr;
    aclrtMalloc(&p, m, ACL_MEM_MALLOC_HUGE_FIRST);
    std::vector<unsigned char> h(m, f);
    aclrtMemcpy(p, m, h.data(), m, ACL_MEMCPY_HOST_TO_DEVICE);
    return p;
}
static aclTensor* Mk(std::vector<int64_t> s, aclDataType dt, void* p)
{
    std::vector<int64_t> st(s.size(), 1);
    for (int i = (int)s.size() - 2; i >= 0; --i) st[i] = s[i + 1] * st[i + 1];
    return aclCreateTensor(s.data(), s.size(), dt, st.data(), 0, ACL_FORMAT_ND, s.data(), s.size(), p);
}
static void Run(const char* tag, int64_t ni, std::vector<int32_t> offs, aclrtStream s)
{
    const int64_t D = 8, NB = (int64_t)offs.size();
    std::vector<float> wh(4 * D);
    for (size_t i = 0; i < wh.size(); ++i) wh[i] = (float)(i + 1);
    void* w = Buf(wh.size() * 4, 0);
    aclrtMemcpy(w, wh.size() * 4, wh.data(), wh.size() * 4, ACL_MEMCPY_HOST_TO_DEVICE);
    void* idx = Buf((size_t)(ni ? ni : 1) * 4, 0);
    if (ni > 0) {
        std::vector<int32_t> ih((size_t)ni);
        for (int64_t i = 0; i < ni; ++i) ih[(size_t)i] = (int32_t)(i % 4);
        aclrtMemcpy(idx, ih.size() * 4, ih.data(), ih.size() * 4, ACL_MEMCPY_HOST_TO_DEVICE);
    }
    void* off = Buf(offs.size() * 4, 0);
    aclrtMemcpy(off, offs.size() * 4, offs.data(), offs.size() * 4, ACL_MEMCPY_HOST_TO_DEVICE);
    void* out = Buf((size_t)(NB * D) * 4, 0x7F);
    void* o2b = Buf((size_t)(ni ? ni : 1) * 4, 0x7F);
    void* bsz = Buf((size_t)NB * 4, 0x7F);
    void* mid = Buf((size_t)NB * 4, 0x7F);
    uint64_t ws = 0; aclOpExecutor* ex = nullptr;
    int g = (int)aclnnEmbeddingBagGetWorkspaceSize(
        Mk({4, D}, ACL_FLOAT, w), Mk({ni}, ACL_INT32, idx), Mk({(int64_t)offs.size()}, ACL_INT32, off),
        false, 0, false, nullptr, false, -1,
        Mk({NB, D}, ACL_FLOAT, out), Mk({ni}, ACL_INT32, o2b), Mk({NB}, ACL_INT32, bsz), Mk({NB}, ACL_INT32, mid),
        &ws, &ex);
    int e = -1;
    if (g == 0 && ex != nullptr) {
        void* wq = nullptr;
        if (ws > 0) aclrtMalloc(&wq, ws, ACL_MEM_MALLOC_HUGE_FIRST);
        e = (int)aclnnEmbeddingBag(wq, ws, ex, s);
        aclrtSynchronizeStream(s);
        if (wq != nullptr) aclrtFree(wq);
    }
    std::vector<float> oh((size_t)(NB * D), 0.0f);
    aclrtMemcpy(oh.data(), oh.size() * 4, out, oh.size() * 4, ACL_MEMCPY_DEVICE_TO_HOST);
    size_t left = 0;
    for (float v : oh) { unsigned int b = 0; std::memcpy(&b, &v, 4); if (b == 0x7F7F7F7Fu) ++left; }
    printf("[%s] ni=%lld nb=%lld g=%d e=%d canary=%zu/%zu first=%.3f\n", tag, (long long)ni, (long long)NB, g, e,
           left, oh.size(), oh.empty() ? 0.0f : oh[0]);
}
int main()
{
    aclInit(nullptr); aclrtSetDevice(0);
    aclrtStream s = nullptr; aclrtCreateStream(&s);
    Run("对照 4indices/2bags", 4, {0, 2}, s);
    Run("缺陷 空indices/1bag", 0, {0}, s);
    Run("缺陷 空indices/2bags", 0, {0, 0}, s);
    aclrtDestroyStream(s); aclrtResetDevice(0); aclFinalize();
    return 0;
}

实测输出（A2 910B2 / CANN 9.3.0）：

[对照 4indices/2bags] ni=4 nb=2 g=0 e=0 canary=0/16 first=10.000
[缺陷 空indices/1bag] ni=0 nb=1 g=0 e=0 canary=8/8 first=339615136492207134461438014706212143104.000
[缺陷 空indices/2bags] ni=0 nb=2 g=0 e=0 canary=16/16 first=339615136492207134461438014706212143104.000

影响

output 是直接消费的 embedding 聚合结果。空 bag 在真实数据里并不罕见（按样本切 bag 时存在某个 bag 无候选的情形），调用方拿到 rc=0 后读到的是未初始化显存，数值随机且随分配变化；本用例 canary 8/8、16/16 全保留，等价于整块输出未定义。

建议

早退分支不要直接返回 SUCCESS：offsets 已确定 output 形状时，用 FillScalar(output, 0, executor) 把输出写 0（与 aclnnAll 的空输入处理一致：:280-286 先 GetTensorWithValueTrue 再返回），或按资料口径返回 ACLNN_ERR_PARAM_INVALID 并在错误码表补充「空输入」这一条。
