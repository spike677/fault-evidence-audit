+def test_fp8_topk_is_independent_of_query_length():
+    """Compare the same causal query in single-row and paired-row vector tasks."""
+    torch_npu.npu.set_device(0)
+    device = "npu"
+    key_length, block_size, heads, head_dim = 5688, 128, 32, 128
+    blocks = (key_length + block_size - 1) // block_size
+
+    query = torch.eye(heads, head_dim).unsqueeze(0).to(torch.float8_e4m3fn).npu()
+    key = torch.zeros(blocks * block_size, head_dim)
+    # Keep 2047 keys above the two candidates that compete for the last TopK slot.
+    key[:2047, 1] = 2
+    key[2047, :3] = 1
+    key[2048, 1] = 0.5
+    key = key.reshape(blocks, block_size, 1, head_dim).to(torch.float8_e4m3fn).npu()
+
+    # The first candidate sums to 1 with even/odd partial sums and to 0
+    # with sequential FP32 accumulation. The second candidate scores 0.5.
+    weights = torch.zeros(1, heads, dtype=torch.bfloat16)
+    weights[0, :3] = torch.tensor([2**24, 1, -(2**24)])
+    weights = weights.npu()
+    query_scale = torch.ones(1, heads, device=device)
+    key_scale = torch.ones(blocks, block_size, 1, device=device)
+    block_table = torch.arange(blocks, dtype=torch.int32, device=device).reshape(1, -1)
+    key_lengths = torch.tensor([key_length], dtype=torch.int32, device=device)
+
+    for query_length in (2, 64, 314, key_length):
+        indices = torch_npu.npu_quant_lightning_indexer(
+            query=query.expand(query_length, -1, -1).contiguous(),
+            key=key,
+            weights=weights.expand(query_length, -1).contiguous(),
+            query_dequant_scale=query_scale.expand(query_length, -1).contiguous(),
+            key_dequant_scale=key_scale,
+            actual_seq_lengths_query=torch.tensor(
+                [query_length], dtype=torch.int32, device=device
+            ),
+            actual_seq_lengths_key=key_lengths,
+            block_table=block_table,
+            query_quant_mode=0,
+            key_quant_mode=0,
+            layout_query="TND",
+            layout_key="PA_BSND",
+            sparse_count=2048,
+            sparse_mode=3,
+        )
+        # The penultimate query has the same causal key range at every length.
